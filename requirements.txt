# Python dependencies for ANNC build/runtime.
#
# Install manually with:
#   pip3 install -r requirements.txt
#
# Or let build.sh --enable-python automatically install the lightweight
# build-time packages (pybind11 and nanobind). They are only needed for the
# optional ANNC Python bindings (default off). TensorFlow is checked but not
# auto-installed because it is large and version-sensitive.

pybind11==2.11.1
nanobind>=2.4,<3
numpy<2
pytest>=8,<9
# TensorFlow is only needed while the TensorFlow addons are built
# (ANNC_ENABLE_TENSORFLOW_ADDON, on by default) or the V2 switch
# ANNC_ENABLE_TF2ATIR_V2 is on (the wheel-isolated pb2stablehlo frontend
# requires 2.20.x). Builds that disable both do not need it.
tensorflow==2.20.0
